\begin{array}{l@{\qquad\qquad}l}
\textbf{Pl. hibás megoldás:} & \textbf{helyes megoldás:} \\[8pt]
\begin{aligned}
x^3 + 2x^2 + x &= 0 \\
&\Downarrow \xleftarrow{:x} \\
x^2 + 2x + 1 &= 0 \\
x &= -1
\end{aligned}
&
\begin{aligned}
x^3 + 2x^2 + x &= 0 \\
x(x^2 + 2x + 1) &= 0 \\
x &= 0 \\
&\text{vagy} \\
x^2 + 2x + 1 = 0 &\Leftrightarrow x = -1
\end{aligned}
\end{array}
